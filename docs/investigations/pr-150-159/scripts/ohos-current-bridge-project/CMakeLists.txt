cmake_minimum_required(VERSION 3.17)
project(erika_current_review_bridges LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(review_root /work)
set(review_runtime /target/aarch64-unknown-linux-ohos/release/liberika_capi.so)
add_library(erika_capi SHARED IMPORTED GLOBAL)
set_target_properties(erika_capi PROPERTIES IMPORTED_LOCATION ${review_runtime})
add_library(erika_flutter SHARED ${review_root}/packages/erika_flutter/ohos/src/main/cpp/erika_flutter_plugin.cpp)
target_include_directories(erika_flutter PRIVATE ${review_root}/packages/erika_flutter/ohos/src/main/cpp/include ${review_root}/packages/erika_flutter/native/include)
add_library(erika_ohos SHARED ${review_root}/packages/erika_ohos/src/main/cpp/erika_ohos.cpp)
target_include_directories(erika_ohos PRIVATE ${review_root}/packages/erika_ohos/src/main/cpp/include)
foreach(bridge erika_flutter erika_ohos)
  target_compile_options(${bridge} PRIVATE -fno-exceptions -ffunction-sections -fdata-sections)
  target_link_options(${bridge} PRIVATE -Wl,--gc-sections)
  target_link_libraries(${bridge} PRIVATE erika_capi libace_napi.z.so libnative_window.so libhilog_ndk.z.so)
endforeach()
